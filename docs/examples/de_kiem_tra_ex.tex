\section*{Phần I. Trắc nghiệm}
\begin{ex}
Giá trị của $1+1$ là
\choice{\True $2$}{$3$}{$4$}{$5$}
\loigiai{Ta có $1+1=2$. Chọn A.}
\end{ex}

\section*{Phần II. Đúng sai}
\begin{ex}
Xét tính đúng sai của các khẳng định sau.
\choiceTF{\True Số $2$ là số chẵn.}{Số $3$ là số chẵn.}
{\True $2+3=5$.}{$2+3=6$.}
\loigiai{Các ý a, c đúng; b, d sai.}
\end{ex}

\section*{Phần III. Trả lời ngắn}
\begin{ex}
Tính $2+3$.
\shortans{5}
\loigiai{$2+3=5$.}
\end{ex}
